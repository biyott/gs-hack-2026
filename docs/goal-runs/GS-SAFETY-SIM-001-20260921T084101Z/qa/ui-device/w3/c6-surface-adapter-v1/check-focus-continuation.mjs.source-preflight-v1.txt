function assertRaw(evidence) {
  const range = evidence.visibleTargetRange, positive = range.right > range.left && range.bottom > range.top;

function assertTraversal(target, events, minimumFrames) {
  assert.equal(target.traversal.complete, true);
  assert.ok(target.frames.length >= minimumFrames);
  assert.equal(target.traversal.ranges.length, target.frames.length);
  const observations = events.filter((event) => event.selector === target.selector && event.evidence);
  assert.ok(observations.length >= target.frames.length * 2);
  const height = observations[0].evidence.targetBounds.height;
  let coveredTo = 0;
  for (const range of target.traversal.ranges) {
    assert.ok(range.top <= coveredTo + 1 && range.bottom > coveredTo + 1, "Geometric traversal gap or stall");
    coveredTo = Math.max(coveredTo, range.bottom);
  }
  assert.ok(coveredTo >= height - 1 && Math.abs(target.traversal.coveredTo - coveredTo) <= 1);
  assert.ok(observations.every(({ evidence }) => evidence.viewport.scrollX === 0 && evidence.viewport.scrollY === 0), "Nested owner must preserve document origin");
  if (minimumFrames > 1) assert.ok(observations.some(({ evidence }) => evidence.ancestors.some((item) => item.label === "div.owner" && item.scrollTop > 0)));
}
